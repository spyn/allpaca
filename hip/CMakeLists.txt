cmake_minimum_required(VERSION 3.21)
project(allpaca_kernels LANGUAGES CXX)

set(ALLPACA_KERNEL_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/kernels.cpp)

if(DEFINED ENV{HIP_PATH})
  file(TO_CMAKE_PATH "$ENV{HIP_PATH}" _HIP_ROOT)
elseif(DEFINED ENV{ROCM_PATH})
  file(TO_CMAKE_PATH "$ENV{ROCM_PATH}" _HIP_ROOT)
endif()

if(_HIP_ROOT)
  list(APPEND CMAKE_PREFIX_PATH "${_HIP_ROOT}")
endif()

find_package(hip QUIET)

if(hip_FOUND)
  enable_language(HIP)
  set_source_files_properties(${ALLPACA_KERNEL_SRC} PROPERTIES LANGUAGE HIP)
  add_library(allpaca_kernels static ${ALLPACA_KERNEL_SRC})
  target_include_directories(allpaca_kernels PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(allpaca_kernels PUBLIC cxx_std_17)
  target_link_libraries(allpaca_kernels PUBLIC hip::device)
  if(NOT DEFINED ENV{AMDGPU_TARGETS} AND NOT DEFINED AMDGPU_TARGETS)
    set(AMDGPU_TARGETS "gfx1200;gfx1201" CACHE STRING "AMD GPU ISAs")
  endif()
  set_property(TARGET allpaca_kernels PROPERTY HIP_ARCHITECTURES ${AMDGPU_TARGETS})
  target_compile_definitions(allpaca_kernels PRIVATE __HIP_PLATFORM_AMD__)
else()
  add_library(allpaca_kernels static ${ALLPACA_KERNEL_SRC})
  target_include_directories(allpaca_kernels PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(allpaca_kernels PUBLIC cxx_std_17)
  message(WARNING "HIP not found; allpaca_kernels is a host stub (NOT_IMPLEMENTED).")
endif()
